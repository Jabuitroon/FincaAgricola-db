-- Insertar producto
CREATE OR REPLACE PROCEDURE procInsertProduct(
    v_nombre        IN VARCHAR2,
    v_descripcion   IN VARCHAR2,
    v_cantidad      IN NUMBER,
    v_precio        IN NUMBER,
    v_img           IN VARCHAR2,
    vfk_proveedor   IN NUMBER,
    vfk_categoria   IN NUMBER,
    v_result        OUT NUMBER
)
IS
BEGIN
    INSERT INTO tbl_producto (
        pro_nombre, pro_descripcion, pro_cantidad, pro_precio, pro_img,
        tbl_proveedor_pro_id, tbl_categoria_cat_id
    )
    VALUES (
        v_nombre, v_descripcion, v_cantidad, v_precio, v_img,
        vfk_proveedor, vfk_categoria
    );

    v_result := SQL%ROWCOUNT;
END procInsertProduct;
/

-- Actualizar producto
CREATE OR REPLACE PROCEDURE procUpdateProduct(
    v_id            IN NUMBER,
    v_nombre        IN VARCHAR2,
    v_descripcion   IN VARCHAR2,
    v_cantidad      IN NUMBER,
    v_precio        IN NUMBER,
    v_img           IN VARCHAR2,
    vfk_proveedor   IN NUMBER,
    vfk_categoria   IN NUMBER,
    v_result        OUT NUMBER
)
IS
BEGIN
    UPDATE tbl_producto
    SET 
        pro_nombre = v_nombre,
        pro_descripcion = v_descripcion,
        pro_cantidad = v_cantidad,
        pro_precio = v_precio,
        pro_img = v_img,
        tbl_proveedor_pro_id = vfk_proveedor,
        tbl_categoria_cat_id = vfk_categoria
    WHERE pro_id = v_id;

    v_result := SQL%ROWCOUNT;
END procUpdateProduct;
/

-- Mostrar productos con detalles
CREATE OR REPLACE PROCEDURE procSelectProduct(
    p_cursor OUT SYS_REFCURSOR
)
IS
BEGIN
    OPEN p_cursor FOR
        SELECT 
            p.pro_id AS producto_id,
            p.pro_nombre,
            p.pro_descripcion,
            p.pro_cantidad,
            p.pro_precio,
            p.pro_img,
            p.tbl_proveedor_pro_id,
            pr.pro_nombre AS proveedor_nombre,
            p.tbl_categoria_cat_id,
            c.cat_nombre
        FROM 
            tbl_producto p
            INNER JOIN tbl_proveedor pr ON p.tbl_proveedor_pro_id = pr.pro_id
            INNER JOIN tbl_categoria c ON p.tbl_categoria_cat_id = c.cat_id;
END procSelectProduct;
/

-- Mostrar productos solo ID y nombre (para DDL)
CREATE OR REPLACE PROCEDURE procSelectProductDDL(
    p_cursor OUT SYS_REFCURSOR
)
IS
BEGIN
    OPEN p_cursor FOR
        SELECT pro_id, pro_nombre FROM tbl_producto;
END procSelectProductDDL;
/

-- Eliminar producto
CREATE OR REPLACE PROCEDURE procDeleteProduct(
    v_id     IN NUMBER,
    v_result OUT NUMBER
)
IS
BEGIN
    DELETE FROM tbl_producto WHERE pro_id = v_id;
    v_result := SQL%ROWCOUNT;
END procDeleteProduct;
/
